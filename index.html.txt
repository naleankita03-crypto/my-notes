📌 Step 2: Create ProductController
using Microsoft.AspNetCore.Mvc;
using MyApp.Models;
using System.Collections.Generic;

namespace MyApp.Controllers
{
    public class ProductController : Controller
    {
        // Sample Data
        private static List<Product> products = new List<Product>()
        {
            new Product { Id = 1, Name = "Laptop", Price = 50000 },
            new Product { Id = 2, Name = "Mobile", Price = 20000 },
            new Product { Id = 3, Name = "Headphones", Price = 3000 }
        };

        // 👉 Action returning View
        public IActionResult Index()
        {
            return View(products); // Sends data to View
        }

        // 👉 Action returning JSON
        public JsonResult GetProducts()
        {
            return Json(products); // Returns JSON data
        }

        // 👉 Action returning single product as JSON
        public JsonResult GetProductById(int id)
        {
            var product = products.Find(p => p.Id == id);
            return Json(product);
        }
    }
}
